The goal is Python a Python programmer would write by hand and happily edit: Owen's "as idiomatic as Python can possibly be".

  • Names follow PEP 8, Python's style guide (Q9):
    • functions, methods, variables and properties in snake_case, lower case joined by underscores: turn_face_up
    • classes in PascalCase, each word capitalized: Card, BankAccount
    • constants in capitals joined by underscores: Card.RANKS
    • a name that clashes with a Python keyword gets a trailing underscore, PEP 8's own rule: class_ (Q14)
  • Native loops, not a core call per item: for number in range(1, 101):.
  • Native operators where the kinds are known:
    • a == b
    • f"{self.rank}-of-{self.suit}"
    • self.suit in ["diamonds", "hearts"]
  • Getters are @property; stored properties are declared in the class body, as dataclasses do.
  • A spell list is a real Python sequence:
    • deck[0], len(deck)
    • for card in deck
    • if not stock: for "the stock is empty"
  • Nothing is None, and asked for as self.rank is None.
  • Type hints wherever spell knows the kind; a number is a float (J37).
  • The core is called only where Python has no word of its own: spell's test lines, a seeded shuffle, a pause.
from __future__ import annotations

def play_fizzbuzz() -> None:
    """File FizzBuzz.spell"""
    for number in range(1, 101):
        if (number / 15).is_integer():
            print(number, "fizzbuzz")
        elif (number / 3).is_integer():
            print(number, "fizz")
        elif (number / 5).is_integer():
            print(number, "buzz")
        else:
            print(number)


play_fizzbuzz()
 
from spell_core import core, Thing, List, prop, h


class Card(Thing):
    """definition of a Card with nice english aliases for working with it"""

    # card ranks
    RANKS = ("ace", 2, 3, 4, 5, 6, 7, 8, 9, 10, "jack", "queen", "king")
    rank = prop(one_of=RANKS)

    # card suits
    SUITS = ("clubs", "diamonds", "hearts", "spades")
    suit = prop(one_of=SUITS)

    @property
    def color(self) -> str:
        """color as derivation of suit"""
        if self.suit in ["diamonds", "hearts"]:
            return "red"
        return "black"

    @property
    def value(self) -> float:
        """value as a derivation of rank"""
        return core.position_of(Card.RANKS, self.rank)

    def is_a_suit(self, suit) -> bool:
        """\"card is a spade", "...is a club" etc"""
        return self.suit == suit

    @property
    def name(self) -> str:
        """name as a derivation of name/suit"""
        return f"{self.rank}-of-{self.suit}"

    def turn_face_up(self) -> None:
        """
        Turn card face up or face down
        Note that this will animate if you `wait for turn the card face up`
        """
        self.direction = "up"
        core.pause_for(50, "msec")
